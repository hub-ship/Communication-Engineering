2.3 运算符优先级要点

  • 逻辑运算符短路特性：`a && b` 中 a 为假时 b 不执行 —— 常用于保护：`if (p != NULL && p->x > 0)`
  • 自增：`i++`（先用后加）vs `++i`（先加后用）
  • 位运算（通信/嵌入式高频！）：`& 与 | 或 ^ 异或 ~ 取反 << 左移 >> 右移`
unsigned char reg = 0xA5;      /* 1010 0101 */
reg |= 0x01;    /* 置位第0位: 1010 0101 | 0000 0001 */
reg &= ~0x02;   /* 清零第1位: 与上 1111 1101 */
reg ^= 0x04;    /* 翻转第2位 */
if (reg & 0x08) {/* 测试第3位是否为1 */}   /* 单片机寄存器操作三大件 */

5.3 经典算法：排序与查找

/* 冒泡排序 */
void bubble(int a[], int n) {
    for (int i = 0; i < n-1; i++)
        for (int j = 0; j < n-1-i; j++)
            if (a[j] > a[j+1]) { int t=a[j]; a[j]=a[j+1]; a[j+1]=t; }
}
/* 折半查找(必须有序) */
int binsearch(int a[], int n, int key) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;   /* 防溢出写法 */
        if (a[mid] == key) return mid;
        a[mid] < key ? (lo = mid+1) : (hi = mid-1);
    }
    return -1;
}

6.2 值传递 —— C函数参数的本质

💡 C语言只有值传递！形参是实参的副本。想在函数内修改外部变量，必须传地址（指针）。这是理解指针意义的钥匙。
void swap_wrong(int a, int b) { int t=a; a=b; b=t; }  /* 换了个寂寞 */
void swap(int *a, int *b) { int t=*a; *a=*b; *b=t; }   /* 正确 */
int x=1, y=2;
swap(&x, &y);      /* x=2, y=1 */

第七章 指针（核心中的核心）

7.1 指针的概念

int a = 10;
int *p = &a;      /* p存储a的地址; *p就是a本身 */

printf("%d %p\n", a, p);     /* 10, 地址 */
printf("%d %p\n", *p, &a);   /* 10, 同一地址 */
*p = 20;                     /* 通过指针修改a → a=20 */
⚠️ 野指针三宗罪：①未初始化就解引用 ②free后继续使用(悬垂指针) ③返回局部变量地址。指针错误是段错误(Segmentation Fault)的头号原因。习惯：`int *p = NULL;` 起手，用前判空。

7.2 指针与数组

int a[5] = {1,2,3,4,5};
int *p = a;              /* 数组名≈首元素地址 */
/* a[i] ≡ *(a+i) ≡ *(p+i) ≡ p[i] 全部等价! */
printf("%d %d %d\n", a[2], *(a+2), *(p+2));
p++;                     /* 指针加1 = 移动一个int(4字节),不是1字节 */
/* sizeof(a)=20(整个数组)  sizeof(p)=8(64位指针)  这是a与p的本质区别 */

7.3 指针与函数（传址调用）

/* 数组作参数自动退化为指针 */
int sum(int arr[], int n) {     /* 等价于 int *arr */
    int s = 0;
    for (int i = 0; i < n; i++) s += arr[i];
    return s;
}
/* 输出型参数: 一级指针改值,二级指针改指针 */
void make_node(struct Node **pp, int v);

7.4 动态内存管理

#include <stdlib.h>
int *arr = (int *)malloc(100 * sizeof(int));  /* 堆上分配,内容随机 */
if (arr == NULL) { /* 分配失败必须检查 */ return -1; }
int *brr = (int *)calloc(100, sizeof(int));   /* 自动清零 */
arr = (int *)realloc(arr, 200 * sizeof(int)); /* 扩容 */
free(arr);  arr = NULL;    /* 用完立即释放并置NULL,防止悬垂 */

7.5 函数指针（回调机制）

int add(int a, int b) { return a + b; }
int sub(int a, int b) { return a - b; }

int apply(int (*op)(int,int), int x, int y) { return op(x, y); }

int main(void) {
    int (*fp)(int,int) = add;         /* 声明函数指针并指向add */
    printf("%d\n", fp(3, 4));          /* 7 */
    printf("%d\n", apply(sub, 10, 4)); /* 6, 回调 */
}

函数指针是 qsort、中断向量表、状态机的灵魂，嵌入式开发天天用。

7.6 指针总结表

声明含义
int *p指向int的指针
int *p[10]指针数组（10个指针）
int (*p)[10]数组指针（指向整行的二维数组用）
int **p二级指针（指针的地址）
int (*p)(int)函数指针
const int *p指向常量的指针（*p不可改，p可改）
int *const p常量指针（p不可改，*p可改）

第十一章 链表与数据结构入门

11.1 单链表的实现

typedef struct node { int data; struct node *next; } node_t;

/* 头插法建表 */
node_t *push_front(node_t *head, int v) {
    node_t *n = malloc(sizeof(node_t));
    n->data = v;  n->next = head;
    return n;                      /* 新节点成为头 */
}
/* 遍历 */
for (node_t *p = head; p != NULL; p = p->next)
    printf("%d -> ", p->data);
/* 删除值为v的节点(注意二级指针/头节点处理) */
node_t *delete(node_t *head, int v) {
    node_t **pp = &head;
    while (*pp && (*pp)->data != v) pp = &(*pp)->next;
    if (*pp) { node_t *t = *pp; *pp = t->next; free(t); }
    return head;
}
/* 释放整表 */
while (head) { node_t *t = head; head = head->next; free(t); }

第十四章 习题与思考

1.（读程序）`int a[5]={1,2,3,4,5}; int *p=a;` 求 *(p+2)、*p++、(*p)++ 各是什么？

*(p+2) = 3（p未变）；
*p++：后缀++优先，先取*p=1，然后p指向a[1]；
(*p)++：对a[0]自增，a[0]变2，p不动。三者的区别是指针运算与解引用优先级的经典考点。

2.（改错）指出下面程序的三处错误。 
char *get_msg(void) {
    char buf[64] = "hello";   /* buf是栈上局部数组 */
    return buf;               /* ①返回局部变量地址,函数返回即失效 */
}
void demo(void) {
    int *p;                   /* ②野指针未初始化 */
    *p = 10;
    char s[8];
    strcpy(s, "0123456789");  /* ③缓冲区溢出,源11字节>目标8字节 */
}

修正：buf 加 static 或由调用者传入；p = malloc 或指向已有变量；用 strncpy/加大缓冲。

3.（编程）不用库函数，统计字符串中大写、小写、数字字符个数。 
void count(const char *s, int *up, int *low, int *dig) {
    *up = *low = *dig = 0;
    for (; *s; s++) {
        if (*s >= 'A' && *s <= 'Z') (*up)++;
        else if (*s >= 'a' && *s <= 'z') (*low)++;
        else if (*s >= '0' && *s <= '9') (*dig)++;
    }
}

要点：用指针遍历到 '\0'；输出参数用指针（因为 C 只有值传递）。

4.（思考）为什么 `sizeof(数组名)` 和 `sizeof(指针)` 结果不同？数组名和指针何时等价？

sizeof(a) 得整个数组字节数，sizeof(p) 得指针本身大小（64位系统为8）。数组名在传参和表达式中退化为首元素指针（&a[0]），例外场景仅有 sizeof 和 &amp;a。理解这一点就不会在函数里 sizeof 形参数组了。

5.（编程）实现函数 `void reverse(char *s)`，原地翻转字符串。 
void reverse(char *s) {
    char *e = s;
    while (*e) e++;          /* e指向'\0' */
    e--;                     /* e指向最后一个字符 */
    while (s < e) {
        char t = *s; *s = *e; *e = t;
        s++; e--;
    }
}

双指针首尾交换，O(n) 时间 O(1) 空间。

6.（思考）结构体中位域和共用体在通信协议开发中各有什么用途？

位域：按比特映射协议字段（如帧头4bit类型+4bit版本），直接读写硬件寄存器/协议位段；
共用体：同一份数据多视角访问 —— union{ unsigned int u; unsigned char b[4]; } 可同时按整数或字节序访问，用于大小端转换、数据序列化。

7.（综合）设计一个函数，把十六进制字符串 "1A2B" 转换为整数 6699。 
unsigned int htoi(const char *s) {
    unsigned int v = 0;
    for (; *s; s++) {
        char c = *s;
        if (c >= '0' && c <= '9')      v = v*16 + (c - '0');
        else if (c >= 'a' && c <= 'f') v = v*16 + (c - 'a' + 10);
        else if (c >= 'A' && c <= 'F') v = v*16 + (c - 'A' + 10);
        else return 0;   /* 非法字符 */
    }
    return v;            /* 1*4096+10*256+2*16+11 = 6699 */
}
